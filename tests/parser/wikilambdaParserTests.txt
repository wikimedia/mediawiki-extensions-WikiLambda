# Parser tests for the WikiLambda parser functions.
#
# The fragment store is empty here, so each {{#function:…}} call gives the pending placeholder.
# The PHPUnit tests for WikifunctionsPFragmentHandler cover the cached, failed and offline results.

!! test
WikiLambda: Function call gives the pending placeholder
!! wikitext
{{#function:Z10000|foo}}
!! html/parsoid
<p><span class="mw-async-not-ready mw-wikifunctions-fragment-pending" typeof="mw:Transclusion mw:ParserFunction/function" about="#mwt1" data-parsoid='{"pi":[[{"k":"1"},{"k":"2"}]]}' data-mw='{"parts":[{"parserfunction":{"target":{"wt":"#function","key":"function"},"params":{"1":{"wt":"Z10000"},"2":{"wt":"foo"}},"i":0}}]}'>⏳ Function is being called…</span></p>
!! end
